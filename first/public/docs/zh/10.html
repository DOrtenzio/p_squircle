10.1 liboqs：参考 C 库

liboqs 是 Open Quantum Safe（OQS）项目开发的参考 C 库，用于后量子密码学。它提供了由 NIST 标准化的 ML-KEM 和其他 PQC 算法的优化且恒定时间的实现。

从源码安装

# 依赖项（Ubuntu/Debian）
sudo apt-get install cmake gcc libssl-dev

# 克隆和编译
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 完整示例：C 语言中使用 ML-KEM-768 的 KEM

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // 初始化算法
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM 不受支持\n");
        return 1;
    }

    printf("算法：%s\n", kem->method_name);
    printf("公钥：%zu 字节\n", kem->length_public_key);
    printf("私钥：%zu 字节\n", kem->length_secret_key);
    printf("密文：%zu 字节\n", kem->length_ciphertext);
    printf("共享密钥：%zu 字节\n\n", kem->length_shared_secret);

    // 分配内存
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen（Alice）
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen 失败\n"); return 1; }
    printf("✓ KeyGen 完成\n");

    // 2. Encaps（Bob：只知道 pk）
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps 失败\n"); return 1; }
    printf("✓ Encaps 完成\n");

    // 3. Decaps（Alice：使用 sk 解封 ct）
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps 失败\n"); return 1; }
    printf("✓ Decaps 完成\n\n");

    // 4. 验证
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ 共享密钥匹配！\n");
        printf("   前几个字节：");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ 错误：共享密钥不匹配！\n");
    }

    // 安全内存清理
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);

    return 0;
}

编译：

gcc -o kem_example kem_example.c -loqs -lcrypto
./kem_example

10.3 将派生密钥与 AES-256-GCM 一起使用

从 ML-KEM 获得共享密钥后，可以将其用作对称密码的密钥。在生产环境中，建议使用 HKDF 派生单独的密钥。

#include <openssl/evp.h>
#include <oqs/oqs.h>

// 从 ML-KEM 获得 ss（32 字节）后：
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // 初始化 AES-256-GCM
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);

    // 使用共享密钥作为密钥
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);

    // 加密
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);

    // 获取认证标签
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);

    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python：liboqs 绑定

liboqs 为 Python 提供了官方绑定，用于快速测试和原型设计。

pip install liboqs-python
import oqs

# 可用算法
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps（Bob 端）
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps（Alice 端）
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ 共享密钥：{shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python：kyber-py（仅用于教学）

纯教育用途，有一个纯 Python 实现，不需要外部 C 库。

警告：kyber-py 不是恒定时间的，不应在生产环境中使用。

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 性能基准测试

在 Intel Core i7-12700K 上的性能基准测试（单核，无 SIMD 指令）：

操作 ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

使用 AVX2 优化

操作 ML-KEM-768（无 SIMD） ML-KEM-768（AVX2） 加速比
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

与 ECDH X25519 的比较

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
交换 ~30 µs ~48 µs
公钥 32 B 1184 B

ML-KEM 大约比 X25519 慢 2 倍，但在大多数实际连接中，TLS 握手延迟开销可以忽略不计（< 1 ms）。

10.7 OpenSSL：安装 OQS 提供程序

OpenSSL 3.x 通过 oqsprovider 提供程序支持 ML-KEM。从 3.4 版本开始，支持是原生的或易于集成的。

# 编译 oqs-provider
git clone https://github.com/open-quantum-safe/oqs-provider.git
cd oqs-provider
cmake -DOPENSSL_ROOT_DIR=/usr/local/ssl .
make
sudo make install

# 验证
openssl list -providers
# 应该显示：oqsprovider

10.8 使用 OpenSSL CLI 生成 ML-KEM 密钥

# 生成 ML-KEM-768 密钥对
openssl genpkey -algorithm mlkem768 -out private.pem
openssl pkey -in private.pem -pubout -out public.pem

# 查看密钥详情
openssl pkey -in private.pem -text -noout

10.9 使用 ML-KEM-768 的 TLS 1.3

本地测试（服务器 + 客户端）

# 服务器
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# 客户端（在另一个终端中）
openssl s_client -connect localhost:4433 -groups mlkem768

检查支持的组

openssl list -key-exchange-algorithms | grep -i kem

10.10 混合策略：X25519 + ML-KEM-768

推荐的迁移方法是 混合化：将 ECDH（X25519）与 ML-KEM-768 结合。最终密钥从两者派生：

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

优点：如果两种方案中至少有一种保持安全，则通信安全。如果量子计算机破解 X25519，ML-KEM 保护；如果 ML-KEM 存在未知缺陷，X25519 保护。

10.11 TLS 1.3 中的混合密码套件

组名 组成
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

从 Chrome 131（2024 年底）开始，X25519MLKEM768 是默认组。Firefox 128+ 支持相同的组。

OpenSSL 配置

# 同时接受经典 X25519 和混合组的服务器
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# 优先使用混合组的客户端
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 使用 OpenSSL 在 C 应用程序中集成

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // 设置首选组：混合组优先，经典组作为后备
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "组配置错误\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 迁移检查清单

对于应用开发者

  • 盘点使用非对称加密的地方（密钥交换、签名、KEM）。
  • 识别哪些数据需要保密超过 5 年。
  • 更新依赖项：OpenSSL ≥ 3.2，liboqs ≥ 0.10。
  • 采用混合方法：TLS 使用 X25519 + ML-KEM-768。
  • 对于数字签名：迁移到 ML-DSA（FIPS 204）或 SLH-DSA（FIPS 205）。
  • 测试协议中密钥和密文大小的增加。
  • 验证超时和缓冲区大小是否假定密钥很小。

对于基础设施管理员

  • 将 TLS 栈更新到支持 ML-KEM 的版本。
  • 在 TLS 服务器上启用 X25519MLKEM768。
  • 监控客户端兼容性（Chrome 131+、Firefox 128+ 支持混合 PQC）。
  • 规划长期密钥轮换。

参考标准

标准 内容
FIPS 203 ML-KEM（密钥封装）
FIPS 204 ML-DSA（基于格的数字签名）
FIPS 205 SLH-DSA（基于哈希的签名，保守型）

10.14 场景：REST API 迁移

当前状况（迁移前）

客户端 → 使用 X25519 的 TLS 1.3 → API 服务器

阶段 1：混合（今天，推荐）

客户端 → 使用 X25519MLKEM768 的 TLS 1.3 → API 服务器

# nginx 配置
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# Apache 配置
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

阶段 2：纯后量子（未来）

客户端 → 使用 ML-KEM-768 + ML-DSA 的 TLS 1.3 → API 服务器

当大多数客户端支持原生 ML-KEM 且服务器证书使用 ML-DSA 签名时采用。
