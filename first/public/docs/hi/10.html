10.1 liboqs: संदर्भ C लाइब्रेरी

liboqs, पोस्ट-क्वांटम क्रिप्टोग्राफी के लिए संदर्भ C लाइब्रेरी है, जिसे Open Quantum Safe (OQS) प्रोजेक्ट द्वारा विकसित किया गया है। यह ML-KEM और NIST द्वारा मानकीकृत अन्य PQC एल्गोरिदम के अनुकूलित और स्थिर-समय (constant-time) कार्यान्वयन प्रदान करता है।

स्रोतों से स्थापना

# निर्भरताएँ (Ubuntu/Debian)
sudo apt-get install cmake gcc libssl-dev

# क्लोन और कंपाइल करें
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 पूर्ण उदाहरण: C में ML-KEM-768 के साथ KEM

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // एल्गोरिदम को इनिशियलाइज़ करें
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM समर्थित नहीं है\n");
        return 1;
    }

    printf("एल्गोरिदम: %s\n", kem->method_name);
    printf("सार्वजनिक कुंजी: %zu बाइट\n", kem->length_public_key);
    printf("निजी कुंजी: %zu बाइट\n", kem->length_secret_key);
    printf("सिफर टेक्स्ट: %zu बाइट\n", kem->length_ciphertext);
    printf("साझा गुप्त कुंजी: %zu बाइट\n\n", kem->length_shared_secret);

    // मेमोरी आवंटन
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen (Alice)
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen विफल\n"); return 1; }
    printf("✓ KeyGen पूर्ण\n");

    // 2. Encaps (Bob: केवल pk जानता है)
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps विफल\n"); return 1; }
    printf("✓ Encaps पूर्ण\n");

    // 3. Decaps (Alice: ct खोलने के लिए sk का उपयोग करता है)
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps विफल\n"); return 1; }
    printf("✓ Decaps पूर्ण\n\n");

    // 4. सत्यापन
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ साझा गुप्त कुंजियाँ मेल खाती हैं!\n");
        printf("   पहले बाइट: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ त्रुटि: साझा गुप्त कुंजियाँ अलग हैं!\n");
    }

    // सुरक्षित मेमोरी सफाई
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);

    return 0;
}

gcc -o kem_example kem_example.c -loqs -lcrypto

10.3 प्राप्त कुंजी को AES-256-GCM के साथ उपयोग करना

ML-KEM से साझा गुप्त कुंजी प्राप्त करने के बाद, इसे एक सममित सिफर के लिए कुंजी के रूप में उपयोग किया जा सकता है। उत्पादन में, अलग कुंजियाँ प्राप्त करने के लिए HKDF का उपयोग करने की सलाह दी जाती है।

#include <openssl/evp.h>
#include <oqs/oqs.h>

// ML-KEM से ss (32 बाइट) प्राप्त करने के बाद:
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // AES-256-GCM को इनिशियलाइज़ करें
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);

    // साझा गुप्त कुंजी को कुंजी के रूप में उपयोग करें
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);

    // एन्क्रिप्शन
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);

    // प्रमाणीकरण टैग प्राप्त करें
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);

    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python: liboqs बाइंडिंग

Liboqs, त्वरित परीक्षण और प्रोटोटाइपिंग के लिए Python में आधिकारिक बाइंडिंग प्रदान करता है।

pip install liboqs-python
import oqs

# उपलब्ध एल्गोरिदम
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps (Bob का पक्ष)
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps (Alice का पक्ष)
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ साझा गुप्त कुंजी: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python: kyber-py (केवल शैक्षणिक)

पूर्णतः शैक्षणिक उद्देश्यों के लिए, एक शुद्ध Python कार्यान्वयन मौजूद है जिसे बाहरी C लाइब्रेरी की आवश्यकता नहीं होती।

ध्यान दें: kyber-py स्थिर-समय (constant-time) नहीं है और इसे उत्पादन में उपयोग नहीं किया जाना चाहिए।

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 प्रदर्शन बेंचमार्क

Intel Core i7-12700K पर बेंचमार्क (एक कोर, बिना SIMD निर्देशों के):

संक्रिया ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

AVX2 अनुकूलन के साथ

संक्रिया ML-KEM-768 (बिना SIMD) ML-KEM-768 (AVX2) स्पीडअप
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

ECDH X25519 से तुलना

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
विनिमय ~30 µs ~48 µs
सार्वजनिक कुंजी 32 B 1184 B

ML-KEM, X25519 से लगभग 2 गुना धीमा है, लेकिन अधिकांश वास्तविक कनेक्शनों में TLS हैंडशेक की विलंबता पर इसका प्रभाव नगण्य होता है (< 1 ms)।

10.7 OpenSSL: OQS प्रोवाइडर की स्थापना

OpenSSL 3.x, oqsprovider प्रोवाइडर के माध्यम से ML-KEM का समर्थन करता है। संस्करण 3.4 से, समर्थन देशी या आसानी से एकीकृत करने योग्य है।

# सत्यापित करें

10.8 OpenSSL CLI के साथ ML-KEM कुंजी उत्पादन

# ML-KEM-768 कुंजी जोड़ी उत्पन्न करें
openssl genpkey -algorithm mlkem768 -out private.pem

# कुंजी का विवरण देखें

10.9 ML-KEM-768 के साथ TLS 1.3

स्थानीय परीक्षण (सर्वर + क्लाइंट)

# सर्वर
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# क्लाइंट (दूसरे टर्मिनल में)
openssl s_client -connect localhost:4433 -groups mlkem768

समर्थित समूहों की जाँच करें

openssl list -key-exchange-algorithms | grep -i kem

10.10 हाइब्रिड रणनीति: X25519 + ML-KEM-768

माइग्रेशन के लिए अनुशंसित दृष्टिकोण हाइब्रिडाइजेशन है: ECDH (X25519) को ML-KEM-768 के साथ जोड़ना। अंतिम कुंजी दोनों से प्राप्त की जाती है:

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

लाभ: यदि दोनों में से कम से कम एक योजना सुरक्षित रहती है, तो यह सुरक्षित है। यदि क्वांटम कंप्यूटर X25519 को तोड़ देता है, तो ML-KEM सुरक्षा प्रदान करता है। यदि ML-KEM में कोई अज्ञात दोष निकलता है, तो X25519 सुरक्षा प्रदान करता है।

10.11 TLS 1.3 में हाइब्रिड सिफर सुइट्स

ग्रुप का नाम संयोजन
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

Chrome 131 (2024 के अंत) से, X25519MLKEM768 डिफ़ॉल्ट ग्रुप है। Firefox 128+ भी इसी ग्रुप का समर्थन करता है।

OpenSSL कॉन्फ़िगरेशन

# सर्वर जो क्लासिक X25519 और हाइब्रिड ग्रुप दोनों को स्वीकार करता है
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# क्लाइंट जो हाइब्रिड ग्रुप को प्राथमिकता देता है
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 OpenSSL के साथ C ऐप्स में एकीकरण

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // पसंदीदा ग्रुप सेट करें: पहले हाइब्रिड, फिर क्लासिक फॉलबैक के रूप में
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "ग्रुप कॉन्फ़िगरेशन में त्रुटि\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 माइग्रेशन के लिए चेकलिस्ट

एप्लिकेशन डेवलपर्स के लिए

  • इन्वेंट्री बनाएं कि असममित क्रिप्टोग्राफी (की एक्सचेंज, सिग्नेचर, KEM) का उपयोग कहाँ किया जा रहा है।
  • पहचानें कि कौन सा डेटा 5 वर्षों से अधिक समय तक गुप्त रहना चाहिए।
  • डिपेंडेंसी अपडेट करें: OpenSSL ≥ 3.2, liboqs ≥ 0.10।
  • हाइब्रिड दृष्टिकोण अपनाएं: TLS के लिए X25519 + ML-KEM-768।
  • डिजिटल सिग्नेचर के लिए: ML-DSA (FIPS 204) या SLH-DSA (FIPS 205) पर माइग्रेट करें।
  • प्रोटोकॉल में कुंजियों और सिफरटेक्स्ट के बढ़े हुए आकार का परीक्षण करें।
  • जांचें कि टाइमआउट और बफर साइज छोटी कुंजियों का अनुमान तो नहीं लगा रहे हैं।

इन्फ्रास्ट्रक्चर मैनेजरों के लिए

  • TLS स्टैक को ML-KEM समर्थन वाले वर्जनों में अपडेट करें।
  • TLS सर्वर पर X25519MLKEM768 सक्षम करें।
  • क्लाइंट संगतता की निगरानी करें (Chrome 131+, Firefox 128+ हाइब्रिड PQC का समर्थन करते हैं)।
  • दीर्घकालिक कुंजी रोटेशन की योजना बनाएं।

संदर्भ मानक

मानक सामग्री
FIPS 203 ML-KEM (Key Encapsulation)
FIPS 204 ML-DSA (लैटिस-आधारित डिजिटल सिग्नेचर)
FIPS 205 SLH-DSA (हैश-आधारित सिग्नेचर, रूढ़िवादी)

10.14 परिदृश्य: REST API का माइग्रेशन

वर्तमान स्थिति (माइग्रेशन से पहले)

क्लाइंट → X25519 के साथ TLS 1.3 → API सर्वर

चरण 1: हाइब्रिड (आज, अनुशंसित)

क्लाइंट → X25519MLKEM768 के साथ TLS 1.3 → API सर्वर

# nginx कॉन्फ़िगरेशन
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# Apache कॉन्फ़िगरेशन
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

चरण 2: विशुद्ध रूप से पोस्ट-क्वांटम (भविष्य)

क्लाइंट → ML-KEM-768 + ML-DSA के साथ TLS 1.3 → API सर्वर

इसे तब अपनाएं जब अधिकांश क्लाइंट मूल रूप से ML-KEM का समर्थन करें और सर्वर प्रमाणपत्र सिग्नेचर के लिए ML-DSA का उपयोग करें।
