10.1 liboqs: مكتبة C المرجعية

liboqs هي مكتبة C المرجعية للتشفير ما بعد الكمي، التي طورها مشروع Open Quantum Safe (OQS). توفر تنفيذات محسّنة وزمنية ثابتة لـ ML-KEM وخوارزميات PQC الأخرى الموحدة من NIST.

التثبيت من المصدر

# التبعيات (أوبونتو/ديبيان)
sudo apt-get install cmake gcc libssl-dev

# الاستنساخ والبناء
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 مثال كامل: KEM مع ML-KEM-768 في C

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // تهيئة الخوارزمية
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM غير مدعوم\n");
        return 1;
    }

    printf("الخوارزمية: %s\n", kem->method_name);
    printf("المفتاح العام: %zu بايت\n", kem->length_public_key);
    printf("المفتاح الخاص: %zu بايت\n", kem->length_secret_key);
    printf("النص المشفر: %zu بايت\n", kem->length_ciphertext);
    printf("السر المشترك: %zu بايت\n\n", kem->length_shared_secret);

    // تخصيص الذاكرة
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen (أليس)
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "فشل KeyGen\n"); return 1; }
    printf("✓ اكتمل KeyGen\n");

    // 2. Encaps (بوب: يعرف فقط pk)
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "فشل Encaps\n"); return 1; }
    printf("✓ اكتمل Encaps\n");

    // 3. Decaps (أليس: تستخدم sk لفتح ct)
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "فشل Decaps\n"); return 1; }
    printf("✓ اكتمل Decaps\n\n");

    // 4. التحقق
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ السر المشترك متطابق!\n");
        printf("   البايتات الأولى: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ خطأ: السر المشترك مختلف!\n");
    }

    // تنظيف آمن للذاكرة
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);
    
    return 0;
}

10.3 استخدام المفتاح المشتق مع AES-256-GCM

بعد الحصول على السر المشترك من ML-KEM، يمكن استخدامه كمفتاح لشيفرة متماثلة. في الإنتاج، يوصى باستخدام HKDF لاشتقاق مفاتيح منفصلة.

#include <openssl/evp.h>
#include <oqs/oqs.h>

// بعد الحصول على ss (32 بايت) من ML-KEM:
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // تهيئة AES-256-GCM
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);
    
    // استخدام السر المشترك كمفتاح
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);
    
    // التشفير
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);
    
    // استرجاع وسم المصادقة
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);
    
    EVP_CIPHER_CTX_free(ctx);
}

10.4 بايثون: روابط liboqs

توفر liboqs روابط بايثون رسمية للاختبار السريع والنمذجة الأولية.

pip install liboqs-python
import oqs

# الخوارزمية المتاحة
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps (جانب بوب)
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps (جانب أليس)
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ السر المشترك: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.6 مقاييس الأداء

مقاييس على Intel Core i7-12700K (نواة واحدة، بدون تعليمات SIMD):

العملية ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

مع تحسينات AVX2

العملية ML-KEM-768 (بدون SIMD) ML-KEM-768 (AVX2) التسريع
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

مقارنة مع ECDH X25519

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
التبادل ~30 µs ~48 µs
المفتاح العام 32 ب 1184 ب

ML-KEM أبطأ بحوالي 2× من X25519، لكن العبء الإضافي على زمن مصافحة TLS لا يُذكر (< 1 مللي ثانية) على معظم الاتصالات الحقيقية.

10.7 OpenSSL: تثبيت مزود OQS

# التحقق
openssl list -providers
# يجب أن يظهر: oqsprovider

10.8 توليد مفاتيح ML-KEM باستخدام سطر أوامر OpenSSL

openssl genpkey -algorithm mlkem768 -out private.pem

# عرض تفاصيل المفتاح
openssl pkey -in private.pem -text -noout

10.9 TLS 1.3 مع ML-KEM-768

اختبار محلي (خادم + عميل)

# الخادم
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# العميل (في طرفية أخرى)
openssl s_client -connect localhost:4433 -groups mlkem768

عرض المجموعات المدعومة

openssl list -key-exchange-algorithms | grep -i kem

10.10 الاستراتيجية الهجينة: X25519 + ML-KEM-768

النهج الموصى به للهجرة هو التهجين: الجمع بين ECDH (X25519) و ML-KEM-768. يُشتق المفتاح النهائي من كليهما:

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

الميزة: آمن إذا صمد واحد على الأقل من المخططين. إذا كسر حاسوب كمومي X25519، يحمي ML-KEM. إذا كان لـ ML-KEM عيب غير معروف، يحمي X25519.

10.11 مجموعات التشفير الهجينة في TLS 1.3

اسم المجموعة التركيب
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

اعتباراً من Chrome 131 (نهاية 2024)، X25519MLKEM768 هي المجموعة الافتراضية. يدعم Firefox 128+ نفس المجموعة.

تكوين OpenSSL

# خادم يقبل كلاً من X25519 الكلاسيكي والمجموعة الهجينة
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# عميل يفضل المجموعة الهجينة
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 التكامل في تطبيقات C مع OpenSSL

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // تعيين المجموعات المفضلة: هجين أولاً، ثم كلاسيكي كاحتياطي
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "خطأ في تكوين المجموعات\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 قائمة تدقيق الهجرة

لمطوري التطبيقات

  • جرد أماكن استخدام التشفير غير المتماثل (تبادل المفاتيح، التوقيع، KEM).
  • تحديد البيانات التي يجب أن تبقى سرية لأكثر من 5 سنوات.
  • تحديث التبعيات: OpenSSL ≥ 3.2، liboqs ≥ 0.10.
  • اعتماد النهج الهجين: X25519 + ML-KEM-768 لـ TLS.
  • للتوقيع الرقمي: الهجرة إلى ML-DSA (FIPS 204) أو SLH-DSA (FIPS 205).
  • اختبار الحجم المتزايد للمفاتيح والنصوص المشفرة في البروتوكولات.
  • التحقق من أن مهلات الوقت وأحجام المخازن المؤقتة لا تفترض مفاتيح صغيرة الحجم.

لمديري البنية التحتية

  • تحديث مكدس TLS إلى إصدارات تدعم ML-KEM.
  • تمكين X25519MLKEM768 على خوادم TLS.
  • مراقبة توافق العملاء (Chrome 131+، Firefox 128+ يدعمون PQC الهجين).
  • التخطيط لتناوب المفاتيح طويلة الأجل.

المعايير المرجعية

المعيار المحتوى
FIPS 203 ML-KEM (تغليف المفاتيح)
FIPS 204 ML-DSA (توقيع رقمي قائم على الشبكات)
FIPS 205 SLH-DSA (توقيع قائم على التجزئة، تحفظي)

10.14 سيناريو: هجرة واجهة برمجة تطبيقات REST

الوضع الحالي (قبل الهجرة)

العميل → TLS 1.3 مع X25519 → خادم API

المرحلة 1: هجين (اليوم، موصى به)

العميل → TLS 1.3 مع X25519MLKEM768 → خادم API

# تكوين nginx
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# تكوين Apache
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

المرحلة 2: ما بعد كمي بحت (المستقبل)

العميل → TLS 1.3 مع ML-KEM-768 + ML-DSA → خادم API

سيتم اعتمادها عندما تدعم غالبية العملاء ML-KEM الأصلي وتستخدم شهادات الخادم ML-DSA للتوقيع.
