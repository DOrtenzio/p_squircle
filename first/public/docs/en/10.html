10.1 liboqs: the reference C library

liboqs is the reference C library for post-quantum cryptography, developed by the Open Quantum Safe (OQS) project. It provides optimized and constant-time implementations of ML-KEM and other NIST-standardized PQC algorithms.

Installation from source

# Dependencies (Ubuntu/Debian)
sudo apt-get install cmake gcc libssl-dev

# Clone and build
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 Complete example: KEM with ML-KEM-768 in C

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // Initialize the algorithm
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM not supported\n");
        return 1;
    }

    printf("Algorithm: %s\n", kem->method_name);
    printf("Public key: %zu bytes\n", kem->length_public_key);
    printf("Private key: %zu bytes\n", kem->length_secret_key);
    printf("Ciphertext: %zu bytes\n", kem->length_ciphertext);
    printf("Shared secret: %zu bytes\n\n", kem->length_shared_secret);

    // Memory allocation
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen (Alice)
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen failed\n"); return 1; }
    printf("✓ KeyGen completed\n");

    // 2. Encaps (Bob: only knows pk)
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps failed\n"); return 1; }
    printf("✓ Encaps completed\n");

    // 3. Decaps (Alice: uses sk to open ct)
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps failed\n"); return 1; }
    printf("✓ Decaps completed\n\n");

    // 4. Verification
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ Shared secrets match!\n");
        printf("   First bytes: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ ERROR: shared secrets differ!\n");
    }

    // Secure memory cleanup
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);
    
    return 0;
}

Compilation:

gcc -o kem_example kem_example.c -loqs -lcrypto
./kem_example

10.3 Using the derived key with AES-256-GCM

After obtaining the shared secret from ML-KEM, it can be used as a key for a symmetric cipher. In production, it is recommended to use HKDF to derive separate keys.

#include <openssl/evp.h>
#include <oqs/oqs.h>

// After obtaining ss (32 bytes) from ML-KEM:
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // Initialize AES-256-GCM
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);
    
    // Use the shared secret as the key
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);
    
    // Encryption
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);
    
    // Retrieve the authentication tag
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);
    
    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python: liboqs bindings

Liboqs provides official Python bindings for rapid testing and prototyping.

pip install liboqs-python
import oqs

# Available algorithm
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps (Bob side)
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps (Alice side)
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ Shared secret: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python: kyber-py (educational only)

For purely educational purposes, there is a pure Python implementation that does not require external C libraries.

Warning: kyber-py is not constant-time and must not be used in production.

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 Performance benchmarks

Benchmarks on Intel Core i7-12700K (single core, no SIMD instructions):

Operation ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

With AVX2 optimizations

Operation ML-KEM-768 (no SIMD) ML-KEM-768 (AVX2) Speedup
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

Comparison with ECDH X25519

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
Exchange ~30 µs ~48 µs
Public key 32 B 1184 B

ML-KEM is about 2× slower than X25519, but the overhead on TLS handshake latency is negligible (< 1 ms) on most real connections.

10.7 OpenSSL: installing the OQS provider

OpenSSL 3.x supports ML-KEM through the oqsprovider provider. Starting from version 3.4, support is native or easily integrable.

# Build oqs-provider
git clone https://github.com/open-quantum-safe/oqs-provider.git
cd oqs-provider
cmake -DOPENSSL_ROOT_DIR=/usr/local/ssl .
make
sudo make install

# Verify
openssl list -providers
# should list: oqsprovider

10.8 ML-KEM key generation with OpenSSL CLI

# Generate ML-KEM-768 key pair
openssl genpkey -algorithm mlkem768 -out private.pem
openssl pkey -in private.pem -pubout -out public.pem

# Display key details
openssl pkey -in private.pem -text -noout

10.9 TLS 1.3 with ML-KEM-768

Local test (server + client)

# Server
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# Client (in another terminal)
openssl s_client -connect localhost:4433 -groups mlkem768

Verify supported groups

openssl list -key-exchange-algorithms | grep -i kem

10.10 Hybrid strategy: X25519 + ML-KEM-768

The recommended approach for migration is hybridization: combining ECDH (X25519) with ML-KEM-768. The final key is derived from both:

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

Advantage: secure if at least one of the two schemes holds. If a quantum computer breaks X25519, ML-KEM protects. If ML-KEM had an unknown flaw, X25519 protects.

10.11 Hybrid cipher suites in TLS 1.3

Group Name Composition
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

Starting from Chrome 131 (late 2024), X25519MLKEM768 is the default group. Firefox 128+ supports the same group.

OpenSSL configuration

# Server that accepts both classic X25519 and the hybrid group
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# Client that prefers the hybrid group
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 Integration in C applications with OpenSSL

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // Set preferred groups: hybrid first, then classics as fallback
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "Error configuring groups\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 Migration checklist

For application developers

  • Inventory where asymmetric cryptography is used (key exchange, signatures, KEM).
  • Identify which data must remain secret for more than 5 years.
  • Update dependencies: OpenSSL ≥ 3.2, liboqs ≥ 0.10.
  • Adopt hybrid approach: X25519 + ML-KEM-768 for TLS.
  • For digital signatures: migrate to ML-DSA (FIPS 204) or SLH-DSA (FIPS 205).
  • Test the increased size of keys and ciphertexts in protocols.
  • Verify that timeouts and buffer sizes do not assume small key sizes.

For infrastructure managers

  • Update TLS stack to versions with ML-KEM support.
  • Enable X25519MLKEM768 on TLS servers.
  • Monitor client compatibility (Chrome 131+, Firefox 128+ support hybrid PQC).
  • Plan long-term key rotation.

Reference standards

Standard Content
FIPS 203 ML-KEM (Key Encapsulation)
FIPS 204 ML-DSA (lattice-based digital signature)
FIPS 205 SLH-DSA (hash-based signature, conservative)

10.14 Scenario: migrating a REST API

Current situation (pre-migration)

Client → TLS 1.3 with X25519 → API Server

Phase 1: hybrid (today, recommended)

Client → TLS 1.3 with X25519MLKEM768 → API Server

# nginx configuration
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# Apache configuration
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

Phase 2: pure post-quantum (future)

Client → TLS 1.3 with ML-KEM-768 + ML-DSA → API Server

To be adopted when the majority of clients support native ML-KEM and server certificates use ML-DSA for signing.
