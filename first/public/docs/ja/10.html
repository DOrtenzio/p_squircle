10.1 liboqs：C言語リファレンスライブラリ

liboqs は、Open Quantum Safe（OQS）プロジェクトによって開発された、耐量子暗号のためのC言語リファレンスライブラリです。ML-KEMおよびNISTによって標準化された他のPQCアルゴリズムの最適化された定数時間実装を提供します。

ソースからのインストール

# 依存関係（Ubuntu/Debian）
sudo apt-get install cmake gcc libssl-dev

# クローンとコンパイル
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 完全な例：C言語でのML-KEM-768によるKEM

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // アルゴリズムの初期化
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEMがサポートされていません\n");
        return 1;
    }

    printf("アルゴリズム: %s\n", kem->method_name);
    printf("公開鍵: %zu バイト\n", kem->length_public_key);
    printf("秘密鍵: %zu バイト\n", kem->length_secret_key);
    printf("暗号文: %zu バイト\n", kem->length_ciphertext);
    printf("共有秘密: %zu バイト\n\n", kem->length_shared_secret);

    // メモリ割り当て
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen（アリス）
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen失敗\n"); return 1; }
    printf("✓ KeyGen完了\n");

    // 2. Encaps（ボブ：pkのみを知っている）
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps失敗\n"); return 1; }
    printf("✓ Encaps完了\n");

    // 3. Decaps（アリス：skを使用してctを開く）
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps失敗\n"); return 1; }
    printf("✓ Decaps完了\n\n");

    // 4. 検証
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ 共有秘密が一致しました！\n");
        printf("   最初のバイト: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ エラー：共有秘密が異なります！\n");
    }

    // 安全なメモリ消去
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);

    return 0;
}

コンパイル：

gcc -o kem_example kem_example.c -loqs -lcrypto
./kem_example

10.3 導出鍵をAES-256-GCMで使用する

ML-KEMから共有秘密を取得した後、それを対称暗号の鍵として使用できます。本番環境では、HKDFを使用して個別の鍵を導出することを推奨します。

#include <openssl/evp.h>
#include <oqs/oqs.h>

// ML-KEMからss（32バイト）を取得した後：
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // AES-256-GCMの初期化
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);

    // 共有秘密を鍵として使用
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);

    // 暗号化
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);

    // 認証タグを取得
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);

    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python：liboqsバインディング

Liboqsは、迅速なテストとプロトタイピングのための公式Pythonバインディングを提供しています。

pip install liboqs-python
import oqs

# 利用可能なアルゴリズム
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps（ボブ側）
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps（アリス側）
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ 共有秘密: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python：kyber-py（教育目的のみ）

純粋に教育目的のために、外部Cライブラリを必要としないPythonのみの実装が存在します。

注意： kyber-py は定数時間ではなく、本番環境で使用してはいけません。

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 パフォーマンスベンチマーク

Intel Core i7-12700K（1コア、SIMD命令なし）でのベンチマーク：

操作 ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

AVX2最適化あり

操作 ML-KEM-768（SIMDなし） ML-KEM-768（AVX2） 高速化
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

ECDH X25519との比較

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
交換 ~30 µs ~48 µs
公開鍵 32 B 1184 B

ML-KEMはX25519より約2倍遅いですが、実際の接続におけるTLSハンドシェイクのレイテンシに対するオーバーヘッド（1 ms未満）はほとんどの場合無視できます。

10.7 OpenSSL：OQSプロバイダのインストール

OpenSSL 3.xは、プロバイダ oqsprovider を通じてML-KEMをサポートします。バージョン3.4以降、サポートはネイティブまたは容易に統合可能です。

# oqs-providerのコンパイル
git clone https://github.com/open-quantum-safe/oqs-provider.git
cd oqs-provider
cmake -DOPENSSL_ROOT_DIR=/usr/local/ssl .
make
sudo make install

# 確認
openssl list -providers
# 表示されるはず：oqsprovider

10.8 OpenSSL CLIでのML-KEM鍵生成

# ML-KEM-768鍵ペアの生成
openssl genpkey -algorithm mlkem768 -out private.pem

10.9 ML-KEM-768を使用したTLS 1.3

ローカルテスト（サーバー + クライアント）

# サーバー
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# クライアント（別のターミナルで）
openssl s_client -connect localhost:4433 -groups mlkem768

サポートされているグループの確認

openssl list -key-exchange-algorithms | grep -i kem

10.10 ハイブリッド戦略：X25519 + ML-KEM-768

移行に推奨されるアプローチはハイブリッド化です。ECDH（X25519）とML-KEM-768を組み合わせます。最終鍵は両方から導出されます。

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

利点： 少なくとも一方のスキームが安全であれば安全です。量子コンピュータがX25519を破ってもML-KEMが保護し、ML-KEMに未知の欠陥があってもX25519が保護します。

10.11 TLS 1.3におけるハイブリッド暗号スイート

グループ名 構成
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

Chrome 131（2024年末）以降、X25519MLKEM768 がデフォルトグループです。Firefox 128+も同じグループをサポートしています。

OpenSSL設定

# クラシックX25519とハイブリッドグループの両方を受け入れるサーバー
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# ハイブリッドグループを優先するクライアント
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 OpenSSLを使用したCアプリケーションへの統合

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // 優先グループの設定：ハイブリッドを先に、クラシックをフォールバックとして
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "グループ設定エラー\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 移行のためのチェックリスト

アプリケーション開発者向け

  • 公開鍵暗号（鍵交換、署名、KEM）が使用されている場所を棚卸しする。
  • 5年以上秘密にしておく必要があるデータを特定する。
  • 依存関係を更新する：OpenSSL ≥ 3.2、liboqs ≥ 0.10。
  • ハイブリッドアプローチを採用する：TLSには X25519 + ML-KEM-768。
  • デジタル署名については、ML-DSA（FIPS 204）またはSLH-DSA（FIPS 205）に移行する。
  • プロトコルにおける鍵と暗号文のサイズ増加をテストする。
  • タイムアウトやバッファサイズが小さな鍵を想定していないことを確認する。

インフラストラクチャ管理者向け

  • TLSスタックをML-KEMサポートバージョンに更新する。
  • TLSサーバーで X25519MLKEM768 を有効にする。
  • クライアントの互換性を監視する（Chrome 131+、Firefox 128+はハイブリッドPQCをサポート）。
  • 長期的な鍵ローテーションを計画する。

参考標準

標準 内容
FIPS 203 ML-KEM（鍵カプセル化）
FIPS 204 ML-DSA（格子ベースデジタル署名）
FIPS 205 SLH-DSA（ハッシュベース署名、保守的）

10.14 シナリオ：REST APIの移行

現在の状況（移行前）

クライアント → TLS 1.3（X25519） → APIサーバー

フェーズ1：ハイブリッド（現在、推奨）

クライアント → TLS 1.3（X25519MLKEM768） → APIサーバー

# nginx設定
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# Apache設定
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

フェーズ2：純粋な耐量子（将来）

クライアント → TLS 1.3（ML-KEM-768 + ML-DSA） → APIサーバー

ほとんどのクライアントがML-KEMをネイティブサポートし、サーバー証明書が署名にML-DSAを使用するようになったときに採用します。
