10.1 liboqs: C 참조 라이브러리

liboqs는 Open Quantum Safe(OQS) 프로젝트에서 개발한 양자 내성 암호화를 위한 C 참조 라이브러리입니다. NIST에서 표준화한 ML-KEM 및 기타 PQC 알고리즘의 최적화되고 일정 시간 구현을 제공합니다.

소스에서 설치

# 의존성 (Ubuntu/Debian)
sudo apt-get install cmake gcc libssl-dev

# 클론 및 컴파일
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 완전한 예제: C에서 ML-KEM-768을 사용한 KEM

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // 알고리즘 초기화
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM이 지원되지 않음\n");
        return 1;
    }

    printf("알고리즘: %s\n", kem->method_name);
    printf("공개 키: %zu 바이트\n", kem->length_public_key);
    printf("개인 키: %zu 바이트\n", kem->length_secret_key);
    printf("암호문: %zu 바이트\n", kem->length_ciphertext);
    printf("공유 비밀: %zu 바이트\n\n", kem->length_shared_secret);

    // 메모리 할당
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen (앨리스)
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen 실패\n"); return 1; }
    printf("✓ KeyGen 완료\n");

    // 2. Encaps (밥: pk만 알고 있음)
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps 실패\n"); return 1; }
    printf("✓ Encaps 완료\n");

    // 3. Decaps (앨리스: sk를 사용하여 ct 열기)
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps 실패\n"); return 1; }
    printf("✓ Decaps 완료\n\n");

    // 4. 검증
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ 공유 비밀이 일치합니다!\n");
        printf("   처음 바이트: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ 오류: 공유 비밀이 다릅니다!\n");
    }

    // 메모리 안전 삭제
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);

    return 0;
}

컴파일:

gcc -o kem_example kem_example.c -loqs -lcrypto
./kem_example

10.3 파생 키를 AES-256-GCM과 함께 사용

ML-KEM에서 공유 비밀을 얻은 후, 이를 대칭 암호의 키로 사용할 수 있습니다. 프로덕션에서는 HKDF를 사용하여 별도의 키를 파생하는 것이 좋습니다.

#include <openssl/evp.h>
#include <oqs/oqs.h>

// ML-KEM에서 ss(32바이트)를 얻은 후:
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // AES-256-GCM 초기화
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);

    // 공유 비밀을 키로 사용
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);

    // 암호화
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);

    // 인증 태그 검색
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);

    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python: liboqs 바인딩

Liboqs는 빠른 테스트와 프로토타이핑을 위한 공식 Python 바인딩을 제공합니다.

pip install liboqs-python
import oqs

# 사용 가능한 알고리즘
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps (밥 측)
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps (앨리스 측)
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ 공유 비밀: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python: kyber-py (교육용)

순수하게 교육 목적으로, 외부 C 라이브러리가 필요 없는 순수 Python 구현이 존재합니다.

주의: kyber-py는 일정 시간이 아니므로 프로덕션에서 사용해서는 안 됩니다.

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 성능 벤치마크

Intel Core i7-12700K (단일 코어, SIMD 명령어 없음) 벤치마크:

연산 ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

AVX2 최적화 사용 시

연산 ML-KEM-768 (SIMD 없음) ML-KEM-768 (AVX2) 속도 향상
KeyGen 37 µs ~10 µs ~3.7배
Encaps 43 µs ~12 µs ~3.6배
Decaps 48 µs ~13 µs ~3.7배

ECDH X25519과의 비교

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
교환 ~30 µs ~48 µs
공개 키 32 B 1184 B

ML-KEM은 X25519보다 약 2배 느리지만, 대부분의 실제 연결에서 TLS 핸드셰이크 지연에 대한 오버헤드는 미미합니다(< 1 ms).

10.7 OpenSSL: OQS 프로바이더 설치

OpenSSL 3.x는 oqsprovider를 통해 ML-KEM을 지원합니다. 버전 3.4부터 지원이 기본 또는 쉽게 통합 가능합니다.

# oqs-provider 컴파일
git clone https://github.com/open-quantum-safe/oqs-provider.git
cd oqs-provider
cmake -DOPENSSL_ROOT_DIR=/usr/local/ssl .
make
sudo make install

# 확인
openssl list -providers
# oqsprovider가 표시되어야 함

10.8 OpenSSL CLI로 ML-KEM 키 생성

# ML-KEM-768 키 쌍 생성
openssl genpkey -algorithm mlkem768 -out private.pem
openssl pkey -in private.pem -pubout -out public.pem

# 키 세부 정보 확인
openssl pkey -in private.pem -text -noout

10.9 ML-KEM-768을 사용한 TLS 1.3

로컬 테스트 (서버 + 클라이언트)

# 서버
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# 클라이언트 (다른 터미널에서)
openssl s_client -connect localhost:4433 -groups mlkem768

지원 그룹 확인

openssl list -key-exchange-algorithms | grep -i kem

10.10 하이브리드 전략: X25519 + ML-KEM-768

마이그레이션을 위해 권장되는 접근 방식은 하이브리드화입니다: ECDH(X25519)와 ML-KEM-768을 결합합니다. 최종 키는 둘 모두에서 파생됩니다:

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

장점: 두 방식 중 적어도 하나가 안전하게 유지되면 보안이 보장됩니다. 양자 컴퓨터가 X25519를 깨면 ML-KEM이 보호하고, ML-KEM에 알려지지 않은 결함이 있으면 X25519가 보호합니다.

10.11 TLS 1.3의 하이브리드 암호 스위트

그룹 이름 구성
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

Chrome 131(2024년 말)부터 X25519MLKEM768이 기본 그룹입니다. Firefox 128+도 동일한 그룹을 지원합니다.

OpenSSL 구성

# 클래식 X25519와 하이브리드 그룹을 모두 허용하는 서버
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# 하이브리드 그룹을 우선하는 클라이언트
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 OpenSSL을 사용한 C 애플리케이션 통합

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // 선호 그룹 설정: 하이브리드 우선, 클래식은 대비책으로
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "그룹 구성 오류\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 마이그레이션 체크리스트

애플리케이션 개발자용

  • 비대칭 암호화(키 교환, 서명, KEM)가 사용되는 위치를 목록화.
  • 5년 이상 비밀로 유지되어야 하는 데이터 식별.
  • 의존성 업데이트: OpenSSL ≥ 3.2, liboqs ≥ 0.10.
  • TLS에 X25519 + ML-KEM-768 하이브리드 접근 방식 채택.
  • 디지털 서명의 경우 ML-DSA(FIPS 204) 또는 SLH-DSA(FIPS 205)로 마이그레이션.
  • 프로토콜에서 증가된 키 및 암호문 크기 테스트.
  • 타임아웃과 버퍼 크기가 작은 키 크기를 가정하지 않는지 확인.

인프라 관리자용

  • ML-KEM을 지원하는 TLS 스택으로 업데이트.
  • TLS 서버에서 X25519MLKEM768 활성화.
  • 클라이언트 호환성 모니터링(Chrome 131+, Firefox 128+는 PQC 하이브리드 지원).
  • 장기 키 교체 계획 수립.

참조 표준

표준 내용
FIPS 203 ML-KEM (키 캡슐화)
FIPS 204 ML-DSA (격자 기반 디지털 서명)
FIPS 205 SLH-DSA (해시 기반 서명, 보수적)

10.14 시나리오: REST API 마이그레이션

현재 상황 (마이그레이션 전)

클라이언트 → TLS 1.3 with X25519 → API 서버

1단계: 하이브리드 (현재, 권장)

클라이언트 → TLS 1.3 with X25519MLKEM768 → API 서버

# nginx 구성
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# Apache 구성
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

2단계: 순수 양자 내성 (미래)

클라이언트 → TLS 1.3 with ML-KEM-768 + ML-DSA → API 서버

대부분의 클라이언트가 ML-KEM을 기본 지원하고 서버 인증서가 ML-DSA를 사용할 때 채택.
