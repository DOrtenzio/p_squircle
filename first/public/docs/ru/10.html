10.1 liboqs: эталонная библиотека C

liboqs — эталонная библиотека C для постквантовой криптографии, разрабатываемая проектом Open Quantum Safe (OQS). Она предоставляет оптимизированные реализации с постоянным временем ML-KEM и других алгоритмов PQC, стандартизированных NIST.

Установка из исходников

# Зависимости (Ubuntu/Debian)
sudo apt-get install cmake gcc libssl-dev

# Клонирование и компиляция
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 Полный пример: KEM с ML-KEM-768 на C

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // Инициализация алгоритма
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM не поддерживается\n");
        return 1;
    }

    printf("Алгоритм: %s\n", kem->method_name);
    printf("Открытый ключ: %zu байт\n", kem->length_public_key);
    printf("Закрытый ключ: %zu байт\n", kem->length_secret_key);
    printf("Шифртекст: %zu байт\n", kem->length_ciphertext);
    printf("Общий секрет: %zu байт\n\n", kem->length_shared_secret);

    // Выделение памяти
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen (Алиса)
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen не удался\n"); return 1; }
    printf("✓ KeyGen завершён\n");

    // 2. Encaps (Боб: знает только pk)
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps не удался\n"); return 1; }
    printf("✓ Encaps завершён\n");

    // 3. Decaps (Алиса: использует sk для открытия ct)
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps не удался\n"); return 1; }
    printf("✓ Decaps завершён\n\n");

    // 4. Проверка
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ Общие секреты совпадают!\n");
        printf("   Первые байты: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ ОШИБКА: общие секреты различаются!\n");
    }

    // Безопасная очистка памяти
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);

    return 0;
}

Компиляция:

gcc -o kem_example kem_example.c -loqs -lcrypto
./kem_example

10.3 Использование производного ключа с AES-256-GCM

После получения общего секрета из ML-KEM его можно использовать как ключ для симметричного шифра. В производстве рекомендуется использовать HKDF для вывода отдельных ключей.

#include <openssl/evp.h>
#include <oqs/oqs.h>

// После получения ss (32 байта) из ML-KEM:
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // Инициализация AES-256-GCM
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);

    // Использование общего секрета как ключа
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);

    // Шифрование
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);

    // Получение тега аутентификации
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);

    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python: биндинг liboqs

Liboqs предоставляет официальные биндинги для Python для быстрого тестирования и прототипирования.

pip install liboqs-python
import oqs

# Доступный алгоритм
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps (сторона Боба)
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps (сторона Алисы)
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ Общий секрет: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python: kyber-py (только для обучения)

Для чисто образовательных целей существует реализация на чистом Python, не требующая внешних библиотек C.

Внимание: kyber-py не имеет постоянного времени и не должен использоваться в производстве.

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 Тесты производительности

Тесты производительности на Intel Core i7-12700K (одно ядро, без инструкций SIMD):

Операция ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

С оптимизациями AVX2

Операция ML-KEM-768 (без SIMD) ML-KEM-768 (AVX2) Ускорение
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

Сравнение с ECDH X25519

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
Обмен ~30 µs ~48 µs
Открытый ключ 32 B 1184 B

ML-KEM примерно в 2 раза медленнее X25519, но задержка рукопожатия TLS является незначительной (< 1 мс) в большинстве реальных соединений.

10.7 OpenSSL: установка провайдера OQS

OpenSSL 3.x поддерживает ML-KEM через провайдер oqsprovider. Начиная с версии 3.4, поддержка является нативной или легко интегрируемой.

# Компиляция oqs-provider
git clone https://github.com/open-quantum-safe/oqs-provider.git
cd oqs-provider
cmake -DOPENSSL_ROOT_DIR=/usr/local/ssl .
make
sudo make install

# Проверка
openssl list -providers
# должен показывать: oqsprovider

10.8 Генерация ключей ML-KEM с OpenSSL CLI

# Генерация пары ключей ML-KEM-768
openssl genpkey -algorithm mlkem768 -out private.pem
openssl pkey -in private.pem -pubout -out public.pem

# Просмотр деталей ключа
openssl pkey -in private.pem -text -noout

10.9 TLS 1.3 с ML-KEM-768

Локальный тест (сервер + клиент)

# Сервер
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# Клиент (в другом терминале)
openssl s_client -connect localhost:4433 -groups mlkem768

Проверка поддерживаемых групп

openssl list -key-exchange-algorithms | grep -i kem

10.10 Гибридная стратегия: X25519 + ML-KEM-768

Рекомендуемым подходом к миграции является гибридизация: комбинация ECDH (X25519) с ML-KEM-768. Финальный ключ выводится из обоих:

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

Преимущество: безопасно, если хотя бы одна из двух схем сохраняется. Если квантовый компьютер взломает X25519, ML-KEM защищает. Если у ML-KEM будет неизвестный дефект, X25519 защищает.

10.11 Гибридные шифр-наборы в TLS 1.3

Название группы Состав
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

Начиная с Chrome 131 (конец 2024), X25519MLKEM768 является группой по умолчанию. Firefox 128+ поддерживает ту же группу.

Конфигурация OpenSSL

# Сервер, принимающий как классический X25519, так и гибридную группу
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# Клиент, предпочитающий гибридную группу
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 Интеграция в приложения на C с OpenSSL

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // Установить предпочтительные группы: гибридная первая, затем классические как запасной вариант
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "Ошибка конфигурации групп\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 Контрольный список для миграции

Для разработчиков приложений

  • Инвентаризация мест использования асимметричной криптографии (обмен ключами, подписи, KEM).
  • Идентификация данных, которые должны оставаться секретными более 5 лет.
  • Обновление зависимостей: OpenSSL ≥ 3.2, liboqs ≥ 0.10.
  • Принятие гибридного подхода: X25519 + ML-KEM-768 для TLS.
  • Для цифровых подписей: миграция на ML-DSA (FIPS 204) или SLH-DSA (FIPS 205).
  • Тестирование увеличения размера ключей и шифртекстов в протоколах.
  • Проверка, что таймауты и размеры буферов не предполагают малых ключей.

Для администраторов инфраструктуры

  • Обновление стека TLS до версий с поддержкой ML-KEM.
  • Включение X25519MLKEM768 на серверах TLS.
  • Мониторинг совместимости клиентов (Chrome 131+, Firefox 128+ поддерживают гибридный PQC).
  • Планирование долгосрочной ротации ключей.

Справочные стандарты

Стандарт Содержание
FIPS 203 ML-KEM (инкапсуляция ключа)
FIPS 204 ML-DSA (цифровая подпись на основе решёток)
FIPS 205 SLH-DSA (подпись на основе хеша, консервативная)

10.14 Сценарий: миграция REST API

Текущая ситуация (до миграции)

Клиент → TLS 1.3 с X25519 → Сервер API

Фаза 1: гибридная (сегодня, рекомендуется)

Клиент → TLS 1.3 с X25519MLKEM768 → Сервер API

# Конфигурация nginx
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# Конфигурация Apache
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

Фаза 2: чисто постквантовая (будущее)

Клиент → TLS 1.3 с ML-KEM-768 + ML-DSA → Сервер API

Для внедрения, когда большинство клиентов будут поддерживать нативный ML-KEM, а сертификаты сервера будут использовать ML-DSA для подписи.
