10.1 liboqs: еталонна бібліотека C

liboqs — це еталонна бібліотека C для постквантової криптографії, що розробляється проектом Open Quantum Safe (OQS). Вона надає оптимізовані реалізації з постійним часом ML-KEM та інших алгоритмів PQC, стандартизованих NIST.

Встановлення з вихідних кодів

# Залежності (Ubuntu/Debian)
sudo apt-get install cmake gcc libssl-dev

# Клонування та компіляція
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 Повний приклад: KEM з ML-KEM-768 на C

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // Ініціалізація алгоритму
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM не підтримується\n");
        return 1;
    }

    printf("Алгоритм: %s\n", kem->method_name);
    printf("Публічний ключ: %zu байт\n", kem->length_public_key);
    printf("Приватний ключ: %zu байт\n", kem->length_secret_key);
    printf("Шифртекст: %zu байт\n", kem->length_ciphertext);
    printf("Спільний секрет: %zu байт\n\n", kem->length_shared_secret);

    // Виділення пам'яті
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen (Аліса)
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen не вдався\n"); return 1; }
    printf("✓ KeyGen завершено\n");

    // 2. Encaps (Боб: знає тільки pk)
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps не вдався\n"); return 1; }
    printf("✓ Encaps завершено\n");

    // 3. Decaps (Аліса: використовує sk для відкриття ct)
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps не вдався\n"); return 1; }
    printf("✓ Decaps завершено\n\n");

    // 4. Перевірка
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ Спільні секрети збігаються!\n");
        printf("   Перші байти: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ ПОМИЛКА: спільні секрети різняться!\n");
    }

    // Безпечне очищення пам'яті
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);

    return 0;
}

Компіляція:

gcc -o kem_example kem_example.c -loqs -lcrypto
./kem_example

10.3 Використання похідного ключа з AES-256-GCM

Після отримання спільного секрету з ML-KEM його можна використовувати як ключ для симетричного шифру. У виробництві рекомендується використовувати HKDF для виведення окремих ключів.

#include <openssl/evp.h>
#include <oqs/oqs.h>

// Після отримання ss (32 байти) з ML-KEM:
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // Ініціалізація AES-256-GCM
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);

    // Використання спільного секрету як ключа
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);

    // Шифрування
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);

    // Отримання тега автентифікації
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);

    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python: зв'язування liboqs

Liboqs надає офіційні зв'язування для Python для швидкого тестування та прототипування.

pip install liboqs-python
import oqs

# Доступний алгоритм
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps (сторона Боба)
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps (сторона Аліси)
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ Спільний секрет: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python: kyber-py (тільки для навчання)

Для суто освітніх цілей існує реалізація на чистому Python, яка не потребує зовнішніх бібліотек C.

Увага: kyber-py не має постійного часу і не повинен використовуватися у виробництві.

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 Тести продуктивності

Тести продуктивності на Intel Core i7-12700K (одне ядро, без інструкцій SIMD):

Операція ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

З оптимізаціями AVX2

Операція ML-KEM-768 (без SIMD) ML-KEM-768 (AVX2) Прискорення
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

Порівняння з ECDH X25519

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
Обмін ~30 µs ~48 µs
Публічний ключ 32 B 1184 B

ML-KEM приблизно в 2 рази повільніший за X25519, але затримка рукостискання TLS є незначною (< 1 мс) у більшості реальних з'єднань.

10.7 OpenSSL: встановлення провайдера OQS

OpenSSL 3.x підтримує ML-KEM через провайдер oqsprovider. Починаючи з версії 3.4, підтримка є нативною або легко інтегрованою.

# Компіляція oqs-provider
git clone https://github.com/open-quantum-safe/oqs-provider.git
cd oqs-provider
cmake -DOPENSSL_ROOT_DIR=/usr/local/ssl .
make
sudo make install

# Перевірка
openssl list -providers
# має показувати: oqsprovider

10.8 Генерація ключів ML-KEM за допомогою OpenSSL CLI

# Генерація пари ключів ML-KEM-768
openssl genpkey -algorithm mlkem768 -out private.pem
openssl pkey -in private.pem -pubout -out public.pem

# Перегляд деталей ключа
openssl pkey -in private.pem -text -noout

10.9 TLS 1.3 з ML-KEM-768

Локальний тест (сервер + клієнт)

# Сервер
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# Клієнт (в іншому терміналі)
openssl s_client -connect localhost:4433 -groups mlkem768

Перевірка підтримуваних груп

openssl list -key-exchange-algorithms | grep -i kem

10.10 Гібридна стратегія: X25519 + ML-KEM-768

Рекомендованим підходом до міграції є гібридизація: комбінація ECDH (X25519) з ML-KEM-768. Фінальний ключ виводиться з обох:

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

Перевага: безпечно, якщо принаймні одна з двох схем зберігається. Якщо квантовий комп'ютер зламає X25519, ML-KEM захищає. Якщо у ML-KEM буде невідомий дефект, X25519 захищає.

10.11 Гібридні шифр-набори в TLS 1.3

Назва групи Склад
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

Починаючи з Chrome 131 (кінець 2024), X25519MLKEM768 є групою за замовчуванням. Firefox 128+ підтримує ту саму групу.

Конфігурація OpenSSL

# Сервер, що приймає як класичний X25519, так і гібридну групу
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# Клієнт, що віддає перевагу гібридній групі
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 Інтеграція в застосунки на C з OpenSSL

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // Встановити бажані групи: спочатку гібридна, потім класичні як резерв
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "Помилка конфігурації груп\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 Контрольний список для міграції

Для розробників застосунків

  • Інвентаризація місць використання асиметричної криптографії (обмін ключами, підписи, KEM).
  • Ідентифікація даних, які повинні залишатися секретними понад 5 років.
  • Оновлення залежностей: OpenSSL ≥ 3.2, liboqs ≥ 0.10.
  • Прийняття гібридного підходу: X25519 + ML-KEM-768 для TLS.
  • Для цифрових підписів: міграція на ML-DSA (FIPS 204) або SLH-DSA (FIPS 205).
  • Тестування збільшення розміру ключів і шифртекстів у протоколах.
  • Перевірка, що таймаути та розміри буферів не припускають малих ключів.

Для адміністраторів інфраструктури

  • Оновлення стеку TLS до версій з підтримкою ML-KEM.
  • Включення X25519MLKEM768 на серверах TLS.
  • Моніторинг сумісності клієнтів (Chrome 131+, Firefox 128+ підтримують гібридний PQC).
  • Планування довгострокової ротації ключів.

Довідкові стандарти

Стандарт Зміст
FIPS 203 ML-KEM (інкапсуляція ключа)
FIPS 204 ML-DSA (цифровий підпис на основі ґраток)
FIPS 205 SLH-DSA (підпис на основі хеша, консервативний)

10.14 Сценарій: міграція REST API

Поточна ситуація (до міграції)

Клієнт → TLS 1.3 з X25519 → Сервер API

Фаза 1: Гібридна (сьогодні, рекомендується)

Клієнт → TLS 1.3 з X25519MLKEM768 → Сервер API

# Конфігурація nginx
ssl_ecdh_curve X25519MLKEM768:x25519:P-256;

# Конфігурація Apache
SSLOpenSSLConfCmd Curves X25519MLKEM768:x25519:P-256;

Фаза 2: Чисто постквантова (майбутнє)

Клієнт → TLS 1.3 з ML-KEM-768 + ML-DSA → Сервер API

Для впровадження, коли більшість клієнтів будуть підтримувати нативний ML-KEM, а сертифікати сервера будуть використовувати ML-DSA для підпису.
