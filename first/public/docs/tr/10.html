10.1 liboqs: referans C kütüphanesi

liboqs, Open Quantum Safe (OQS) projesi tarafından geliştirilen post-kuantum kriptografi için referans C kütüphanesidir. NIST tarafından standartlaştırılmış ML-KEM ve diğer PQC algoritmalarının optimize edilmiş ve sabit süreli uygulamalarını sağlar.

Kaynaktan kurulum

# Bağımlılıklar (Ubuntu/Debian)
sudo apt-get install cmake gcc libssl-dev

# Klonlama ve derleme
git clone https://github.com/open-quantum-safe/liboqs.git
cd liboqs
mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..
make -j$(nproc)
sudo make install

10.2 Tam örnek: C'de ML-KEM-768 ile KEM

#include <stdio.h>
#include <string.h>
#include <oqs/oqs.h>

int main() {
    OQS_STATUS rc;
    const char *kem_alg = OQS_KEM_alg_ml_kem_768;

    // Algoritmayı başlat
    OQS_KEM *kem = OQS_KEM_new(kem_alg);
    if (kem == NULL) {
        fprintf(stderr, "KEM desteklenmiyor\n");
        return 1;
    }

    printf("Algoritma: %s\n", kem->method_name);
    printf("Genel anahtar: %zu bayt\n", kem->length_public_key);
    printf("Özel anahtar: %zu bayt\n", kem->length_secret_key);
    printf("Şifreli metin: %zu bayt\n", kem->length_ciphertext);
    printf("Paylaşılan sır: %zu bayt\n\n", kem->length_shared_secret);

    // Bellek tahsisi
    uint8_t *pk = malloc(kem->length_public_key);
    uint8_t *sk = malloc(kem->length_secret_key);
    uint8_t *ct = malloc(kem->length_ciphertext);
    uint8_t *ss_enc = malloc(kem->length_shared_secret);
    uint8_t *ss_dec = malloc(kem->length_shared_secret);

    // 1. KeyGen (Alice)
    rc = OQS_KEM_keypair(kem, pk, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "KeyGen başarısız\n"); return 1; }
    printf("✓ KeyGen tamamlandı\n");

    // 2. Encaps (Bob: sadece pk'yi bilir)
    rc = OQS_KEM_encaps(kem, ct, ss_enc, pk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Encaps başarısız\n"); return 1; }
    printf("✓ Encaps tamamlandı\n");

    // 3. Decaps (Alice: ct'yi açmak için sk'yi kullanır)
    rc = OQS_KEM_decaps(kem, ss_dec, ct, sk);
    if (rc != OQS_SUCCESS) { fprintf(stderr, "Decaps başarısız\n"); return 1; }
    printf("✓ Decaps tamamlandı\n\n");

    // 4. Doğrulama
    if (memcmp(ss_enc, ss_dec, kem->length_shared_secret) == 0) {
        printf("✅ Paylaşılan sırlar eşleşiyor!\n");
        printf("   İlk baytlar: ");
        for (int i = 0; i < 8; i++) printf("%02x", ss_enc[i]);
        printf("...\n");
    } else {
        printf("❌ HATA: paylaşılan sırlar farklı!\n");
    }

    // Güvenli bellek temizliği
    OQS_MEM_cleanse(sk, kem->length_secret_key);
    OQS_MEM_cleanse(ss_enc, kem->length_shared_secret);
    OQS_MEM_cleanse(ss_dec, kem->length_shared_secret);
    free(pk); free(ct);
    OQS_KEM_free(kem);

    return 0;
}

Derleme:

gcc -o kem_example kem_example.c -loqs -lcrypto
./kem_example

10.3 Türetilen anahtarı AES-256-GCM ile kullanmak

ML-KEM'den paylaşılan sır elde edildikten sonra, simetrik bir şifre için anahtar olarak kullanılabilir. Üretimde, ayrı anahtarlar türetmek için HKDF kullanılması önerilir.

#include <openssl/evp.h>
#include <oqs/oqs.h>

// ML-KEM'den ss (32 bayt) aldıktan sonra:
void encrypt_with_kem_key(const uint8_t *ss, const uint8_t *plaintext,
                           size_t pt_len, uint8_t *ciphertext, uint8_t *tag,
                           const uint8_t *nonce) {
    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
    int len;

    // AES-256-GCM başlat
    EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL);
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN, 12, NULL);

    // Paylaşılan sırrı anahtar olarak kullan
    EVP_EncryptInit_ex(ctx, NULL, NULL, ss, nonce);

    // Şifreleme
    EVP_EncryptUpdate(ctx, ciphertext, &len, plaintext, pt_len);
    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);

    // Kimlik doğrulama etiketini al
    EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, 16, tag);

    EVP_CIPHER_CTX_free(ctx);
}

10.4 Python: liboqs bağlantısı

Liboqs, hızlı test ve prototip oluşturma için resmi Python bağlantıları sağlar.

pip install liboqs-python
import oqs

# Mevcut algoritma
kem = oqs.KeyEncapsulation("ML-KEM-768")

# KeyGen
public_key = kem.generate_keypair()

# Encaps (Bob tarafı)
kem_bob = oqs.KeyEncapsulation("ML-KEM-768")
ciphertext, shared_secret_enc = kem_bob.encap_secret(public_key)

# Decaps (Alice tarafı)
shared_secret_dec = kem.decap_secret(ciphertext)

assert shared_secret_enc == shared_secret_dec
print(f"✓ Paylaşılan sır: {shared_secret_enc.hex()[:16]}...")

kem.free()
kem_bob.free()

10.5 Python: kyber-py (sadece eğitim amaçlı)

Tamamen eğitim amaçları için, harici C kütüphaneleri gerektirmeyen saf Python uygulaması mevcuttur.

Uyarı: kyber-py sabit süreli değildir ve üretimde kullanılmamalıdır.

                    
                        from kyber import Kyber768
                        kem = Kyber768()
                        pk, sk = kem.keygen()
                        key, ciphertext = kem.enc(pk)
                        _key = kem.dec(sk, ciphertext)
                        assert key == _key

10.6 Performans kıyaslamaları

Intel Core i7-12700K üzerinde performans kıyaslamaları (tek çekirdek, SIMD talimatları yok):

İşlem ML-KEM-512 ML-KEM-768 ML-KEM-1024
KeyGen 24 µs 37 µs 54 µs
Encaps 28 µs 43 µs 62 µs
Decaps 32 µs 48 µs 70 µs

AVX2 optimizasyonları ile

İşlem ML-KEM-768 (SIMD yok) ML-KEM-768 (AVX2) Hızlanma
KeyGen 37 µs ~10 µs ~3.7×
Encaps 43 µs ~12 µs ~3.6×
Decaps 48 µs ~13 µs ~3.7×

ECDH X25519 ile karşılaştırma

X25519 ML-KEM-768
KeyGen ~14 µs ~37 µs
Değişim ~30 µs ~48 µs
Genel anahtar 32 B 1184 B

ML-KEM, X25519'ten yaklaşık 2 kat daha yavaştır, ancak TLS el sıkışma gecikmesindeki ek yük çoğu gerçek bağlantıda ihmal edilebilir düzeydedir (< 1 ms).

10.7 OpenSSL: OQS sağlayıcısının kurulumu

OpenSSL 3.x, oqsprovider sağlayıcısı aracılığıyla ML-KEM'i destekler. 3.4 sürümünden itibaren destek yereldir veya kolayca entegre edilebilir.

# oqs-provider derleme
git clone https://github.com/open-quantum-safe/oqs-provider.git
cd oqs-provider
cmake -DOPENSSL_ROOT_DIR=/usr/local/ssl .
make
sudo make install

# Doğrulama
openssl list -providers
# şunu göstermeli: oqsprovider

10.8 OpenSSL CLI ile ML-KEM anahtarı oluşturma

# ML-KEM-768 anahtar çifti oluştur
openssl genpkey -algorithm mlkem768 -out private.pem
openssl pkey -in private.pem -pubout -out public.pem

# Anahtar ayrıntılarını görüntüle
openssl pkey -in private.pem -text -noout

10.9 ML-KEM-768 ile TLS 1.3

Yerel test (sunucu + istemci)

# Sunucu
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups mlkem768

# İstemci (başka bir terminalde)
openssl s_client -connect localhost:4433 -groups mlkem768

Desteklenen grupları kontrol et

openssl list -key-exchange-algorithms | grep -i kem

10.10 Hibrit strateji: X25519 + ML-KEM-768

Göç için önerilen yaklaşım hibritleştirmedir: ECDH'yi (X25519) ML-KEM-768 ile birleştirmek. Nihai anahtar her ikisinden de türetilir:

\[ K = \text{HKDF}(K_{\text{X25519}} \| K_{\text{ML-KEM}}) \]

Avantaj: iki şemadan en az biri ayakta kalırsa güvenlidir. Bir kuantum bilgisayar X25519'i kırarsa, ML-KEM korur. ML-KEM'in bilinmeyen bir kusuru varsa, X25519 korur.

10.11 TLS 1.3'te hibrit şifre paketleri

Grup Adı Bileşim
X25519MLKEM768 X25519 + ML-KEM-768
SecP256r1MLKEM768 P-256 + ML-KEM-768

Chrome 131'den itibaren (2024 sonu), X25519MLKEM768 varsayılan gruptur. Firefox 128+ aynı grubu destekler.

OpenSSL yapılandırması

# Hem klasik X25519 hem de hibrit grubu kabul eden sunucu
openssl s_server -accept 4433 -cert server.crt -key server.key \
    -groups X25519MLKEM768:x25519:P-256

# Hibrit grubu tercih eden istemci
openssl s_client -connect localhost:4433 \
    -groups X25519MLKEM768:x25519

10.12 OpenSSL ile C uygulamalarında entegrasyon

#include <openssl/ssl.h>
#include <openssl/err.h>

SSL_CTX *create_pqc_context(void) {
    SSL_CTX *ctx = SSL_CTX_new(TLS_server_method());
    if (!ctx) {
        ERR_print_errors_fp(stderr);
        return NULL;
    }

    // Tercih edilen grupları ayarla: önce hibrit, sonra klasik yedek
    if (SSL_CTX_set1_groups_list(ctx, "X25519MLKEM768:x25519:P-256") != 1) {
        fprintf(stderr, "Grup yapılandırma hatası\n");
        SSL_CTX_free(ctx);
        return NULL;
    }

    return ctx;
}

10.13 Göç için kontrol listesi

Uygulama geliştiricileri için

  • Asimetrik kriptografinin (anahtar değişimi, imza, KEM) kullanıldığı yerleri envanterleyin.
  • 5 yıldan fazla gizli kalması gereken verileri belirleyin.
  • Bağımlılıkları güncelleyin: OpenSSL ≥ 3.2, liboqs ≥ 0.10.
  • Hibrit yaklaşımı benimseyin: TLS için X25519 + ML-KEM-768.
  • Dijital imza için: ML-DSA'ya (FIPS 204) veya SLH-DSA'ya (FIPS 205) geçin.
  • Protokollerde anahtar ve şifreli metin boyutlarının artışını test edin.
  • Zaman aşımlarının ve tampon boyutlarının küçük anahtarları varsaymadığını doğrulayın.

Altyapı yöneticileri için

  • TLS yığınını ML-KEM desteği olan sürümlere güncelleyin.
  • TLS sunucularında X25519MLKEM768'i etkinleştirin.
  • İstemci uyumluluğunu izleyin (Chrome 131+, Firefox 128+ hibrit PQC'yi destekler).
  • Uzun vadeli anahtar rotasyonunu planlayın.

Referans standartlar

Standart İçerik
FIPS 203 ML-KEM (Anahtar Kapsülleme)
FIPS 204 ML-DSA (kafes tabanlı dijital imza)
FIPS 205 SLH-DSA (hash tabanlı imza, muhafazakâr)
